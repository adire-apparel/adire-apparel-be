func (h ProductHandler) handleMultipartForm(ctx *gin.Context) {
	if err := ctx.Request.ParseMultipartForm(8 << 20); err != nil {
		ctx.JSON(http.StatusBadRequest, gin.H{
			"message": "Failed to parse multipart form: " + err.Error(),
		})
		return
	}

	name := ctx.PostForm("name")
	description := ctx.PostForm("description")
	categoryId := ctx.PostForm("category_id")
	baseStr := ctx.PostForm("price.base")
	currencyStr := ctx.PostForm("price.currency")
	taxRateStr := ctx.PostForm("price.tax_rate")
	discountPercentageStr := ctx.PostForm("price.discount_percentage")

	form, err := ctx.MultipartForm()
	if err != nil {
		ctx.JSON(http.StatusBadRequest, gin.H{
			"message": "Failed to get multipart form: " + err.Error(),
		})
		return
	}

	files := form.File["images"]
	log.Printf("Number of files: %d", len(files))

	if name == "" || description == "" || categoryId == "" {
		ctx.JSON(http.StatusBadRequest, gin.H{
			"message": "Name, description, and category_id are required",
		})
		return
	}

	price, err := ParsePrice(baseStr, currencyStr, taxRateStr, discountPercentageStr)
	if err != nil {
		ctx.JSON(http.StatusBadRequest, gin.H{
			"message": "Invalid price format: " + err.Error(),
		})
		return
	}
	if err = ValidatePrice(price); err != nil {
		ctx.JSON(http.StatusBadRequest, gin.H{
			"message": err.Error(),
			"status":  http.StatusBadRequest,
			"data":    nil,
		})
		return
	}

	variantsStr := ctx.PostForm("variants")
	log.Println(variantsStr)
	var variants []models.VariantModel
	if err = json.Unmarshal([]byte(variantsStr), &variants); err != nil {
		ctx.JSON(http.StatusBadRequest, gin.H{
			"message": "Invalid variants format: " + err.Error(),
		})
		return
	}
	if err = ValidateVariants(variants); err != nil {
		ctx.JSON(http.StatusBadRequest, gin.H{
			"message": err.Error(),
			"status":  http.StatusBadRequest,
			"data":    nil,
		})
		return
	}

	var imageUrls []string
	if len(files) > 0 {
		imageUrls, err = lib.ImageUploader(files, "products")
		if err != nil {
			log.Printf("Image upload error: %v", err)
			ctx.JSON(http.StatusInternalServerError, gin.H{
				"message": "Failed to upload images: " + err.Error(),
			})
			return
		}
	}

	product := dto.CreateProductPayload{
		Name:        name,
		Description: description,
		CategoryId:  categoryId,
		Images:      imageUrls,
		Price:       price,
		Variants:    variants,
	}

	saved, err := h.service.CreateProduct(product)
	if err != nil {
		ctx.JSON(http.StatusInternalServerError, gin.H{
			"message": "Failed to create product: " + err.Error(),
		})
		return
	}

	ctx.JSON(http.StatusCreated, gin.H{
		"message": "Product created successfully",
		"status":  http.StatusCreated,
		"data":    saved,
	})
}
